97 void resizeEvent(QResizeEvent* event) override {
98 QWidget::resizeEvent(event);
99 // These pages live in a QStackedWidget, which does not honour a
100 // word-wrapped label's height-for-width: the hint ends up clipped to
101 // whatever height was left over. Pin its minimum to what the text
102 // actually needs at the width it got.
103 if (m_explanation != nullptr && m_explanation->width() > 0) {
104 m_explanation->setMinimumHeight(
105 m_explanation->heightForWidth(m_explanation->width()));
106 }
107 }

127 void onTextChanged(const QString& text) {
128 const QString trimmed = text.trimmed();
129 const bool correct = (trimmed == m_expected);
130
131 if (trimmed.isEmpty()) {
132 m_feedback->clear();
133 } else if (correct) {
134 m_feedback->setText(
135 tr("✔ Respuesta correcta, pasa a la siguiente pregunta"));
136 } else {
137 m_feedback->setText(tr("✘ Incorrecto, revisa el enunciado. "
138 "Consulta los estados que has "
139 "construido."));
140 }
141
142 if (m_isComplete == correct) {
143 return;
144 }
145
146 m_isComplete = correct;
147 emit completionChanged(m_isComplete);
148 }
149

151 QString m_title;
152 QString m_expected;
153 QLabel* m_feedback;
154 QLabel* m_explanation = nullptr;
155 QLineEdit* m_edit;
156 bool m_isComplete =
157 false;
