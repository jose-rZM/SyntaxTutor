87 void addGrammar(const QVector<Rule>& rules, bool numbered) {
88 addSection(tr("Gramática"));
89 html_ += QStringLiteral("<table class='grammar' cellspacing='0' "
90 "cellpadding='1'>");
91 for (int i = 0; i < rules.size(); ++i) {
92 html_ += QStringLiteral("<tr>");
93 if (numbered) {
94 html_ += QStringLiteral("<td class='num'>(%1)</td>").arg(i);
95 }
96 html_ += QStringLiteral("<td class='mono'>%1 → %2</td></tr>")
97 .arg(rules[i].first.toHtmlEscaped(),
98 QStringList(rules[i].second.begin(),
99 rules[i].second.end())
100 .join(QLatin1Char(' '))
101 .toHtmlEscaped());
102 }
103 html_ += QStringLiteral("</table>");
104 }

107 void addConversation(const QVector<Message>& messages) {
108 addSection(tr("Conversación"));
109 html_ += QStringLiteral("<table class='chat' width='100%' "
110 "cellspacing='0' cellpadding='6'>");
111 for (const Message& message : messages) {
112 const bool wrong = message.fromStudent && !message.correct;
113 const QString rowClass =
114 wrong ? QStringLiteral("wrong")
115 : (message.fromStudent ? QStringLiteral("student")
116 : QStringLiteral("tutor"));
117 QString who = message.fromStudent ? tr("Usuario") : tr("Tutor");
118 if (wrong) {
119 who += QStringLiteral("<br><span class='flag'>%1</span>")
120 .arg(tr("incorrecta"));
121 }
122 html_ += QStringLiteral("<tr class='%1'><td class='who' "
123 "width='14%'>%2</td><td>%3</td></tr>")
124 .arg(rowClass, who, multiline(message.text));
125 }
126 html_ += QStringLiteral("</table>");
127 }

135 void addResult(const QString& grade, bool passed,
136 const QStringList& lines) {
137 const QString tone =
138 passed ? QStringLiteral("pass") : QStringLiteral("fail");
139 html_ += QStringLiteral("<table class='result' cellspacing='0' "
140 "cellpadding='0'><tr><td class='grade %1'>"
141 "%2</td><td>")
142 .arg(tone, grade.toHtmlEscaped());
143 for (int i = 0; i < lines.size(); ++i) {
144 const bool last = i == lines.size() - 1;
145 html_ += QStringLiteral("<p class='%1'>%2</p>")
146 .arg(last ? QStringLiteral("verdict ") + tone
147 : QStringLiteral("summary"),
148 lines[i].toHtmlEscaped());
149 }
150 html_ += QStringLiteral("</td></tr></table>");
151 }

154 void addReview(const QVector<ExamRecord>& records) {
155 addSection(tr("Revisión"));
156 html_ += QStringLiteral(
157 "<table class='review' width='100%' cellspacing='0' "
158 "cellpadding='5'><thead><tr><th width='40%'>%1</th>"
159 "<th width='27%'>%2</th><th width='27%'>%3</th>"
160 "<th width='6%'></th></tr></thead><tbody>")
161 .arg(tr("Pregunta"), tr("Tu respuesta"),
162 tr("Respuesta correcta"));
163 for (const ExamRecord& record : records) {
164 const QString answer = record.userAnswer.trimmed().isEmpty()
165 ? tr("(vacía)")
166 : record.userAnswer;
167 html_ += QStringLiteral(
168 "<tr class='%1'><td>%2</td><td class='mono'>%3</td>"
169 "<td class='mono'>%4</td><td class='mark'>%5</td>"
170 "</tr>")
171 .arg(record.correct ? QStringLiteral("right")
172 : QStringLiteral("wrong"),
173 multiline(record.question), multiline(answer),
174 multiline(record.correctAnswer),
175 record.correct ? QStringLiteral("✔")
176 : QStringLiteral("✘"));
177 }
178 html_ += QStringLiteral("</tbody></table>");
179 }

224 void addTable(const QStringList& headers,
225 const QVector<QStringList>& rows) {
226 const int dataColumns = static_cast<int>(headers.size()) - 1;
227 for (int first = 0; first < qMax(dataColumns, 1);
228 first += kMaxTableColumns) {
229 const int last = qMin(first + kMaxTableColumns, dataColumns);
230 if (dataColumns > kMaxTableColumns) {
231 html_ += QStringLiteral("<p class='caption'>%1</p>")
232 .arg(tr("Columnas %1 a %2 de %3")
233 .arg(first + 1)
234 .arg(last)
235 .arg(dataColumns));
236 }
237 html_ += QStringLiteral("<table class='grid' cellspacing='0' "
238 "cellpadding='4'><thead><tr>");
239 html_ += QStringLiteral("<th>%1</th>")
240 .arg(headers.value(0).toHtmlEscaped());
241 for (int c = first; c < last; ++c) {
242 html_ += QStringLiteral("<th class='mono'>%1</th>")
243 .arg(headers.value(c + 1).toHtmlEscaped());
244 }
245 html_ += QStringLiteral("</tr></thead><tbody>");
246 for (int r = 0; r < rows.size(); ++r) {
247 html_ += QStringLiteral("<tr class='%1'>")
248 .arg(r % 2 ? QStringLiteral("odd")
249 : QStringLiteral("even"));
250 html_ += QStringLiteral("<th class='mono'>%1</th>")
251 .arg(rows[r].value(0).toHtmlEscaped());
252 for (int c = first; c < last; ++c) {
253 html_ += QStringLiteral("<td class='mono'>%1</td>")
254 .arg(rows[r].value(c + 1).toHtmlEscaped());
255 }
256 html_ += QStringLiteral("</tr>");
257 }
258 html_ += QStringLiteral("</tbody></table>");
259 }
260 }

266 void print(const QString& filePath) const {
267 ensureMonospaceFont();
268 QTextDocument doc;
269 // A fixed size: paper does not follow the on-screen text scale.
270 QFont font = AppTypography::systemFont();
271 font.setPointSizeF(10.0);
272 doc.setDefaultFont(font);
273 doc.setHtml(html());
274
275 QPrinter printer(QPrinter::HighResolution);
276 printer.setOutputFormat(QPrinter::PdfFormat);
277 printer.setOutputFileName(filePath);
278 printer.setPageSize(QPageSize(QPageSize::A4));
279 printer.setPageMargins(QMarginsF(15, 15, 15, 15),
280 QPageLayout::Millimeter);
281 doc.print(&printer);
282 }

290 static constexpr auto kMonospaceFamily = "JetBrains Mono";
291
292 // The PDF names its monospaced font in CSS, which needs a real family:
293 // the platform's fixed font is an internal family on macOS that the
294 // printer cannot resolve. The bundled JetBrains Mono is the same
295 // everywhere; main() only loads it on Windows and Linux.
296 static void ensureMonospaceFont() {
297 if (!QFontDatabase::families().contains(
298 QLatin1StringView(kMonospaceFamily))) {
299 QFontDatabase::addApplicationFont(
300 QStringLiteral(":/resources/fonts/JetBrainsMono-Regular.ttf"));
301 QFontDatabase::addApplicationFont(
302 QStringLiteral(":/resources/fonts/JetBrainsMono-Bold.ttf"));
303 }
304 }
