98inline qreal basePoints(Role role) {
99 for (const Token& token : kTokens) {
100 if (token.role == role) {
101 return token.points;
102 }

150inline QString resolveStyleSheet(const QString& styleSheet,
151 QStringList* unresolved = nullptr) {
152 static const QRegularExpression tokenRe(
153 QStringLiteral("\\$font-([a-z]+(?:-[a-z]+)*)"));
154
155 QString out;
156 qsizetype last = 0;
157 auto it = tokenRe.globalMatch(styleSheet);
158 while (it.hasNext()) {
159 const QRegularExpressionMatch match = it.next();
160 out += styleSheet.mid(last, match.capturedStart() - last);
161 last = match.capturedEnd();
162
163 const QString name = match.captured(1);
164 bool found = false;
165 for (const Token& token : kTokens) {
166 if (name == token.name) {
167 out += formatPoints(token.points * AppTextScale::factor());
168 found = true;
169 break;
170 }
171 }
172 if (!found) {
173 out += match.captured(0);
174 if (unresolved != nullptr) {
175 unresolved->append(name);
176 }
177 }
178 }
179 out += styleSheet.mid(last);
180 return out;
181}

202inline int lengthForText(int designPx) {
203 const QFontInfo rendered(QGuiApplication::font());
204 const qreal textPixels = rendered.pixelSize() > 0
205 ? rendered.pixelSize()
206 : basePoints(Role::Body);
207 return qRound(designPx * textPixels / basePoints(Role::Body));
208}
