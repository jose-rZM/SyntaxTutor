102 void resizeEvent(QResizeEvent* event) override {
103 QWidget::resizeEvent(event);
104 // These pages live in a QStackedWidget, which does not honour a
105 // word-wrapped label's height-for-width: the hint ends up clipped to

136 void onTextChanged(const QString& text) {
137 const QString trimmed = text.trimmed();
138 const bool correct = (compact(trimmed) == compact(m_expected));
139
140 if (trimmed.isEmpty()) {
141 m_feedback->clear();
142 } else if (correct) {
143 m_feedback->setText(
144 tr("✔ Respuesta correcta, pasa a la siguiente pregunta"));
145 } else {
146 m_feedback->setText(tr("✘ Incorrecto, repasa los símbolos "
147 "directores (SD) en el panel del tutor."));
148 }
149
150 if (m_isComplete == correct) {
151 return;
152 }
153
154 m_isComplete = correct;
155 emit completionChanged(m_isComplete);
156 }
157
158 private:
159 static QString compact(const QString& text) {
160 QString result = text;
161 result.remove(QRegularExpression("\\s+"));
162 return result;
163 }
