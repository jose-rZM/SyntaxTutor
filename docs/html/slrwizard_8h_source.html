140 const int nTerm = parser.gr_.st_.terminals_.size();
141 for (int i = 0; i < rawTable.size(); ++i) {
142 for (int j = 0; j < colHeaders.size(); ++j) {
143 const QString sym = colHeaders[j];
144 QString expected;
145 QString explanation;
146
147 if (j < nTerm) {
148 auto itAct = parser.actions_.at(i).find(sym.toStdString());
149 SLR1Parser::s_action act =
150 (itAct != parser.actions_.at(i).end()
151 ? itAct->second
152 : SLR1Parser::s_action{nullptr,
153 SLR1Parser::Action::Empty});
154
155 switch (act.action) {
156 case SLR1Parser::Action::Shift: {
157 const unsigned to =
158 parser.transitions_.at(i).at(sym.toStdString());
159 expected = QString("s%1").arg(to);
160 explanation = tr("Estado %1: existe transición δ(%1, "
161 "'%2'). ¿A qué estado harías shift?")
162 .arg(i)
163 .arg(sym);
164 break;
165 }
166 case SLR1Parser::Action::Reduce: {
167 int idx = -1;
168 for (int k = 0; k < sortedGrammar.size(); ++k) {
169 auto& rule = sortedGrammar[k];
170 if (rule.first.toStdString() ==
171 act.item->antecedent_ &&
172 stdVectorToQVector(act.item->consequent_) ==
173 rule.second) {
174 idx = k;
175 break;
176 }
177 }
178 expected = QString("r%1").arg(idx);
179 explanation = tr("Estado %1: contiene el ítem [%2 → "
180 "...·] y '%3' ∈ SIG(%2). ¿Qué regla "
181 "usas para reducir (0, 1, ...)?")
182 .arg(i)
183 .arg(QString::fromStdString(
184 act.item->antecedent_))
185 .arg(colHeaders[j]);
186 break;
187 }
188 case SLR1Parser::Action::Accept:
189 expected = "acc";
190 explanation = tr("Estado %1: contiene [S → A · $]. "
191 "¿Qué palabra clave usas para "
192 "aceptar?")
193 .arg(i);
194 break;
195 case SLR1Parser::Action::Empty:
196 default:
197 continue;
198 }
199 } else {
200 const auto nonT = sym.toStdString();
201 if (!parser.transitions_.contains(i)) {
202 continue;
203 }
204
205 auto itGo = parser.transitions_.at(i).find(nonT);
206 if (itGo == parser.transitions_.at(i).end()) {
207 continue;
208 }
209
210 expected = QString::number(itGo->second);
211 explanation = tr("Estado %1: δ(%1, '%2') existe. ¿A qué "
212 "estado va la transición? (pon solo el "
213 "número)")
214 .arg(i)
215 .arg(sym);
216 }
