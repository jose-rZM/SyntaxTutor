40class HeightForWidthKeeper : public QObject {
41 public:
42 explicit HeightForWidthKeeper(QWidget* widget)
43 : QObject(widget), widget_(widget) {
44 widget_->installEventFilter(this);
45 }
46
47 protected:
48 bool eventFilter(QObject* watched, QEvent* event) override {
49 if (watched == widget_ && (event->type() == QEvent::Resize ||
50 event->type() == QEvent::LayoutRequest ||
51 event->type() == QEvent::Show)) {
52 update();
53 }
54 return false;
55 }
56
57 private:
58 void update() {
59 QLayout* layout = widget_->layout();
60 if (layout == nullptr || !layout->hasHeightForWidth()) {
61 return;
62 }
63 // The minimum, not the preferred height: the preferred one also
64 // counts what spacers would like to have, which over-constrains.
65 const int needed =
66 layout->totalMinimumHeightForWidth(widget_->width());
67 if (needed <= 0) {
68 return;
69 }
70 if (widget_->minimumHeight() != needed) {
71 widget_->setMinimumHeight(needed);
72 }
73 // A window is not resized by its minimum alone once it is shown.
74 if (widget_->isWindow() && widget_->height() < needed) {
75 widget_->resize(widget_->width(), needed);
76 }
77 }
78
79 QWidget* widget_;
80};

91 new HeightForWidthKeeper(widget);
